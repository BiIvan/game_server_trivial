cmake_minimum_required(VERSION 3.20)
project(game_server LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
list(APPEND CMAKE_PREFIX_PATH "${CMAKE_BINARY_DIR}")
find_package(Boost CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_library(game_model STATIC
    src/model.cpp src/loot_generator.cpp src/collision_detector.cpp)
target_include_directories(game_model PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
add_executable(game_server
    src/main.cpp src/http_server.cpp src/boost_json.cpp
    src/json_loader.cpp src/request_handler.cpp)
target_link_libraries(game_server PRIVATE
    game_model Boost::headers Boost::log_setup Boost::log Threads::Threads)
if(WIN32)
    target_compile_definitions(game_server PRIVATE
        NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    target_link_libraries(game_server PRIVATE ws2_32 mswsock)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(game_model_checks verification/game_model_checks.cpp)
    target_link_libraries(game_model_checks PRIVATE game_model)
    add_test(NAME game_model_checks COMMAND game_model_checks)
endif()
